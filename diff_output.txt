+// =============================================================================
+// FORWARD — calls library functions in sequence, all state explicit

+SubRoutine.ATTN_Forward {
+    fwd_dmod  = TFConfig.DModel
+    fwd_seqln = TFConfig.SeqLen
+    fwd_vocab = TFConfig.VocabSize
+    fwd_scale = TF_ISqrt(fwd_dmod)
+    fwd_tok   = TFState.tokens

+    TFLayer_Embed(TFW.emb, fwd_tok, TFW.Embed, TFW.PosEmb, fwd_seqln, fwd_dmod)
+
+    TFProjCarrier.Wk    = TFW.Wk
+    TFProjCarrier.Wv    = TFW.Wv
+    TFProjCarrier.seqln = fwd_seqln
+    TFProjCarrier.dmod  = fwd_dmod
+    TFLayer_ProjQKV(TFW.qvec, TFW.kvec, TFW.vvec, TFW.emb, TFW.Wq)
+
+    TFLayer_AttnScores(TFW.scores, TFW.qvec, TFW.kvec, fwd_seqln, fwd_dmod, fwd_scale)
+
+    TFLayer_AttnSoftmax(TFW.attn, TFW.scores, fwd_seqln)
+
+    TFLayer_Context(TFW.ctx, TFW.attn, TFW.vvec, fwd_seqln, fwd_dmod)
+
+    TFLayer_Residual(TFW.resid, TFW.ctx, TFW.emb, fwd_seqln, fwd_dmod)
+
+    TFOutProjCarrier.seqln = fwd_seqln
+    TFOutProjCarrier.dmod  = fwd_dmod
+    TFOutProjCarrier.vocab = fwd_vocab
+    TFLayer_OutputProj(TFW.logits, TFW.probs, TFW.resid, TFW.Wout)
+}

+// =============================================================================
+// GENERATE SAMPLE — fills TFState.tokens and TFState.targets
+// =============================================================================
+SubRoutine.ATTN_GenSample {
+    gen_seqln = TFConfig.SeqLen
+    gen_vocab = TFConfig.VocabSize
+    gen_tok   = TFState.tokens
+    gen_tgt   = TFState.targets
+
+    gen_pos = 0
+    WhileLoop LessThan(gen_pos, gen_seqln) {
+        gen_rval = TF_RandMod(gen_vocab)
+        TF_VecSet(gen_tok, gen_pos, gen_rval)
+        gen_pos = Add(gen_pos, 1)
+    }
+    // Reversal task: target = reversed input
+    gen_pos = 0
+    WhileLoop LessThan(gen_pos, gen_seqln) {
+        gen_src = Subtract(Subtract(gen_seqln, 1), gen_pos)
+        gen_val = TF_VecGet(gen_tok, gen_src)
+        TF_VecSet(gen_tgt, gen_pos, gen_val)
+        gen_pos = Add(gen_pos, 1)
+    }
+}

+// =============================================================================
+// EXP LOOKUP — single entry
+// idx clamped to [0, 255], returns Q8 value
+// =============================================================================
+Function.TFAct_Exp {
+    Input:  idx: Integer
+    Output: Integer
+    Body: {
+        safe = TF_Clamp(idx, 0, 255)
+        exp_off = Multiply(TFElem.Size, safe)
+        exp_addr = Add(TFActState.exp_tbl, exp_off)
+        ReturnValue(Dereference(exp_addr))
+    }
+}

+// =============================================================================
+// SOFTMAX
+// dst[i] = exp(src[i] - max) / sum(exp(src[j] - max))
+// Uses EXPTBL. Input and output are integer vectors (Q8 scale).
+// Steps:
+//   1. Find max for numerical stability
+//   2. For each i: diff = max - src[i], idx = diff / 8, dst[i] = EXPTBL[idx]
+//   3. Sum all dst values
+//   4. Divide each by sum (normalize)
+// =============================================================================
+Function.TFAct_Softmax {
+    Input: dst: Integer
+    Input: src: Integer
+    Input: len: Integer
+    Body: {
+        // Step 1: find max
+        maxv = TF_VecMax(src, len)
+
+        // Step 2: compute exp values via table
+        i = 0
+        WhileLoop LessThan(i, len) {
+            diff = Subtract(maxv, TF_VecGet(src, i))
+            idx  = TF_Clamp(Divide(diff, 8), 0, 255)
+            TF_VecSet(dst, i, TFAct_Exp(idx))
+            i = Add(i, 1)
+        }
+
+        // Step 3: sum
+        total = TF_VecSum(dst, len)
+        IfCondition EqualTo(total, 0) ThenBlock: { total = 1 }
+
+        // Step 4: normalize — keep in Q8 range (multiply by 256, divide by sum)
+        i = 0
+        WhileLoop LessThan(i, len) {
+            v = TF_VecGet(dst, i)
+            TF_VecSet(dst, i, Divide(Multiply(v, 256), total))
+            i = Add(i, 1)
+        }
+        ReturnValue(0)
+    }
+}

+// =============================================================================
+// CROSS-ENTROPY LOSS (for reporting only, every LogEvery steps)
+// loss = -ln(p_target) averaged over sequence positions
+// p_target = softmax[target_token], looked up in LOGTBL
+// Returns Q12 fixed-point value
+// =============================================================================
+Function.TFAct_CrossEntropy {
+    Input:  probs:   Integer   // softmax output vector [VocabSize]
+    Input:  target:  Integer   // target token index
+    Output: Integer
+    Body: {
+        // Get probability of target token (Q8, range 0-256)
+        pval = TF_VecGet(probs, target)
+        pval = TF_Clamp(pval, 1, 256)
+        // Look up -ln(p/256) in log table
+        log_off = Multiply(TFElem.Size, pval)
+        log_addr = Add(TFActState.log_tbl, log_off)
+        loss = Dereference(log_addr)
+        ReturnValue(loss)
+    }
+}

+// =============================================================================
+// SOFTMAX GRADIENT
+// For softmax + cross-entropy pair the gradient simplifies beautifully:
+//   dL[i] = softmax[i] - one_hot(target)[i]
+// No log, no division. Result is Q8 shifted to Q15 for backward pass.
+// =============================================================================
+Function.TFAct_SoftmaxGrad {
+    Input: dst:     Integer   // output gradient vector [VocabSize]
+    Input: probs:   Integer   // softmax probabilities [VocabSize]
+    Input: target:  Integer   // target token index
+    Input: vocabsz: Integer
+    Body: {
+        i = 0
+        WhileLoop LessThan(i, vocabsz) {
+            pv = TF_VecGet(probs, i)
+            // Subtract 256 (=1.0 in Q8) from target position
+            IfCondition EqualTo(i, target) ThenBlock: {
+                pv = Subtract(pv, 256)
+            }
+            // Keep gradients in Q8 range (no shift to Q15 which was overflowing)
+            TF_VecSet(dst, i, pv)
+            i = Add(i, 1)
+        }
+        ReturnValue(0)
+    }
+}

+// =============================================================================
+// FLOAT DOT PRODUCT - Using SSE2 float math
+// sum(a[i] * b[i]) - uses float accumulation for precision, returns integer
+// =============================================================================
+Function.TF_FDot {
+    Input:  fdot_vecA: Integer
+    Input:  fdot_vecB: Integer
+    Input:  fdot_len:  Integer
+    Output: Integer
+    Body: {
+        fdot_sum = Float_FromInt(0)
+        fdot_i = 0
+        WhileLoop LessThan(fdot_i, fdot_len) {
+            fdot_va = Float_FromInt(TF_VecGet(fdot_vecA, fdot_i))
+            fdot_vb = Float_FromInt(TF_VecGet(fdot_vecB, fdot_i))
+            fdot_prod = Float_Mul(fdot_va, fdot_vb)
+            fdot_sum = Float_Add(fdot_sum, fdot_prod)
+            fdot_i = Add(fdot_i, 1)
+        }
+        // Convert back to integer for backward pass compatibility (round)
+        ReturnValue(Float_Round(fdot_sum))
+    }
+}

+// =============================================================================
+// FLOAT OUTER ACCUMULATE - Using SSE2 float math
+// grad[r][c] += a[r] * b[c]
+// Uses float internally for precision, stores result as integer
+// =============================================================================
+Function.TF_FOuterAccum {
+    Input:  facc_mat:   Integer
+    Input:  facc_vecA:  Integer
+    Input:  facc_vecB:  Integer
+    Input:  facc_rows:  Integer
+    Input:  facc_cols:  Integer
+    Output: Integer
+    Body: {
+        facc_r = 0
+        WhileLoop LessThan(facc_r, facc_rows) {
+            facc_va = Float_FromInt(TF_VecGet(facc_vecA, facc_r))
+            facc_c = 0
+            WhileLoop LessThan(facc_c, facc_cols) {
+                facc_vb = Float_FromInt(TF_VecGet(facc_vecB, facc_c))
+                facc_old = Float_FromInt(TF_MatGet(facc_mat, facc_r, facc_c, facc_cols))
+                // Float math: accumulate product
+                facc_prod = Float_Mul(facc_va, facc_vb)
+                facc_new = Float_Add(facc_old, facc_prod)
+                // Store as integer
+                TF_MatSet(facc_mat, facc_r, facc_c, facc_cols, Float_ToInt(facc_new))
+                facc_c = Add(facc_c, 1)
+            }
+            facc_r = Add(facc_r, 1)
+        }
+        ReturnValue(1)
+    }

+// FLOAT SGD - Using SSE2 float math
+// w -= lr * grad  where lr = lr_num / lr_den
+// Weight matrix (fsgd_mat) stores raw IEEE 754 float bits (float masters).
+// Gradient matrix (fsgd_grad) stores integers (backward pass output).
+// Updates accumulate in float — no rounding back to integer.
+// =============================================================================
+Function.TF_FSGD {
+    Input:  fsgd_mat:    Integer
+    Input:  fsgd_grad:   Integer
+    Input:  fsgd_rows:   Integer
+    Input:  fsgd_cols:   Integer
+    Input:  fsgd_lr_num: Integer
+    Input:  fsgd_lr_den: Integer
+    Output: Integer
+    Body: {
+        // Compute learning rate as float once
+        fsgd_lr = Float_Div(Float_FromInt(fsgd_lr_num), Float_FromInt(fsgd_lr_den))
+
+        fsgd_r = 0
+        WhileLoop LessThan(fsgd_r, fsgd_rows) {
+            fsgd_c = 0
+            WhileLoop LessThan(fsgd_c, fsgd_cols) {
+                // Gradient is integer, convert to float
+                fsgd_g_i = TF_MatGet(fsgd_grad, fsgd_r, fsgd_c, fsgd_cols)
+                fsgd_g_f = Float_FromInt(fsgd_g_i)
+
+                // Weight is raw float bits, read directly
+                fsgd_w_off = TF_Offset(fsgd_r, fsgd_c, fsgd_cols)
+                fsgd_w_f = Dereference(Add(fsgd_mat, fsgd_w_off))
+
+                // Float math: update = grad * lr, new_w = w - update
+                fsgd_upd = Float_Mul(fsgd_g_f, fsgd_lr)
+                fsgd_new_f = Float_Sub(fsgd_w_f, fsgd_upd)
+
+                // Store float bits directly — no rounding to integer
+                StoreValue(Add(fsgd_mat, fsgd_w_off), fsgd_new_f)
+                fsgd_c = Add(fsgd_c, 1)
+            }
+            fsgd_r = Add(fsgd_r, 1)
+        }
+        ReturnValue(1)
+    }

+        pos = 0
+        WhileLoop LessThan(pos, seqln) {
+            tgt = TF_VecGet(targets, pos)
+            col = 0
+            WhileLoop LessThan(col, vocab) {
+                TF_VecSet(prob_row, col, TF_MatGet(probs, pos, col, vocab))
+                col = Add(col, 1)
+            }
+            TFAct_SoftmaxGrad(dlogit_row, prob_row, tgt, vocab)
+            col = 0
+            WhileLoop LessThan(col, vocab) {
+                TF_MatSet(dLogits, pos, col, vocab, TF_VecGet(dlogit_row, col))
+                col = Add(col, 1)
+            }
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                TF_VecSet(resid_row, col, TF_MatGet(resid, pos, col, dmod))
+                col = Add(col, 1)
+            }
+            TF_FOuterAccum(dWout, dlogit_row, resid_row, vocab, dmod)
+            TF_MatTVecMul(dresid_row, Wout, dlogit_row, vocab, dmod)
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                TF_MatSet(dResid, pos, col, dmod, TF_VecGet(dresid_row, col))
+                col = Add(col, 1)
+            }
+            pos = Add(pos, 1)
+        }
+
+        TF_VecFree(dlogit_row, vocab)
+        TF_VecFree(dresid_row, dmod)
+        TF_VecFree(prob_row,   vocab)
+        TF_VecFree(resid_row,  dmod)
+        ReturnValue(1)
+    }
+}

+// =============================================================================
+// RESIDUAL SPLIT + dV + dScores
+// Reads from TFBackCtx: .attn .vvec .seqln .dmod .scale
+// =============================================================================
+Function.TFTrain_BackAttnCtx {
+    Input:  dCtx:    Integer
+    Input:  dEmb:    Integer
+    Input:  dV:      Integer
+    Input:  dScores: Integer
+    Input:  dResid:  Integer
+    Output: Integer
+    Body: {
+        attn  = TFBackCtx.attn
+        vvec  = TFBackCtx.vvec
+        seqln = TFBackCtx.seqln
+        dmod  = TFBackCtx.dmod
+        scale = TFBackCtx.scale
+
+        dctx_row = TF_VecAlloc(dmod)
+        vrow     = TF_VecAlloc(dmod)
+        dv_accum = TF_VecAlloc(dmod)
+        DebugAssert(NotEqual(dctx_row, 0), "TFTrain_BackAttnCtx: alloc failed")
+
+        // Residual split
+        pos = 0
+        WhileLoop LessThan(pos, seqln) {
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                dv = TF_MatGet(dResid, pos, col, dmod)
+                TF_MatSet(dCtx, pos, col, dmod, dv)
+                TF_MatSet(dEmb, pos, col, dmod, dv)
+                col = Add(col, 1)
+            }
+            pos = Add(pos, 1)
+        }
+
+        // dV - using float accumulation for precision
+        vpos = 0
+        WhileLoop LessThan(vpos, seqln) {
+            TF_VecZero(dv_accum, dmod)
+            ipos = 0
+            WhileLoop LessThan(ipos, seqln) {
+                aw_f = Float_FromInt(TF_MatGet(attn, ipos, vpos, seqln))
+                col = 0
+                WhileLoop LessThan(col, dmod) {
+                    dc_f     = Float_FromInt(TF_MatGet(dCtx, ipos, col, dmod))
+                    old_f    = Float_FromInt(TF_VecGet(dv_accum, col))
+                    prod_f   = Float_Mul(aw_f, dc_f)
+                    scaled_f = Float_Div(prod_f, Float_FromInt(256))
+                    nuw_f    = Float_Add(old_f, scaled_f)
+                    TF_VecSet(dv_accum, col, Float_ToInt(nuw_f))
+                    col = Add(col, 1)
+                }
+                ipos = Add(ipos, 1)
+            }
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                TF_MatSet(dV, vpos, col, dmod, TF_VecGet(dv_accum, col))
+                col = Add(col, 1)
+            }
+            vpos = Add(vpos, 1)
+        }
+
+        // dScores - softmax Jacobian (NOT sigmoid a*(1-a))
+        // For each row i:
+        //   raw[j] = dot(dCtx[i], V[j]) / scale
+        //   wsum   = sum_k( attn[i,k] * raw[k] ) / 256
+        //   dS[i,j] = attn[i,j] * (raw[j] - wsum) / 256
+        f256 = Float_FromInt(256)
+        raw_buf = TF_VecAlloc(seqln)
+        ipos = 0
+        WhileLoop LessThan(ipos, seqln) {
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                TF_VecSet(dctx_row, col, TF_MatGet(dCtx, ipos, col, dmod))
+                col = Add(col, 1)
+            }
+            // Pass 1: compute raw[j] = dot(dCtx[i], V[j]) / scale for all j
+            wsum_f = Float_FromInt(0)
+            jpos = 0
+            WhileLoop LessThan(jpos, seqln) {
+                col = 0
+                WhileLoop LessThan(col, dmod) {
+                    TF_VecSet(vrow, col, TF_MatGet(vvec, jpos, col, dmod))
+                    col = Add(col, 1)
+                }
+                raw_f = Float_Div(Float_FromInt(TF_FDot(dctx_row, vrow, dmod)), Float_FromInt(Multiply(256, scale)))
+                TF_VecSet(raw_buf, jpos, Float_ToInt(raw_f))
+                // Accumulate weighted sum: wsum += attn[i,j] * raw[j] / 256
+                aw_f = Float_FromInt(TF_MatGet(attn, ipos, jpos, seqln))
+                wsum_f = Float_Add(wsum_f, Float_Div(Float_Mul(aw_f, raw_f), f256))
+                jpos = Add(jpos, 1)
+            }
+            // Pass 2: dS[i,j] = attn[i,j] * (raw[j] - wsum) / 256
+            jpos = 0
+            WhileLoop LessThan(jpos, seqln) {
+                aw_f  = Float_FromInt(TF_MatGet(attn, ipos, jpos, seqln))
+                raw_f = Float_FromInt(TF_VecGet(raw_buf, jpos))
+                diff_f = Float_Sub(raw_f, wsum_f)
+                dsc_f  = Float_Div(Float_Mul(aw_f, diff_f), f256)
+                TF_MatSet(dScores, ipos, jpos, seqln, Float_ToInt(dsc_f))
+                jpos = Add(jpos, 1)
+            }
+            ipos = Add(ipos, 1)
+        }
+        TF_VecFree(raw_buf, seqln)
+
+        TF_VecFree(dctx_row, dmod)
+        TF_VecFree(vrow,     dmod)
+        TF_VecFree(dv_accum, dmod)
+        ReturnValue(1)
+    }
+}

+// =============================================================================
+// dQ dK from dScores
+// Reads from TFBackQK: .seqln .dmod
+// =============================================================================
+Function.TFTrain_BackQK {
+    Input:  dQ:      Integer
+    Input:  dK:      Integer
+    Input:  dScores: Integer
+    Input:  qvec:    Integer
+    Input:  kvec:    Integer
+    Output: Integer
+    Body: {
+        seqln = TFBackQK.seqln
+        dmod  = TFBackQK.dmod
+
+        qrow = TF_VecAlloc(dmod)
+        DebugAssert(NotEqual(qrow, 0), "TFTrain_BackQK: alloc failed")
+
+        TF_MatZero(dQ, seqln, dmod)
+        TF_MatZero(dK, seqln, dmod)
+
+        ipos = 0
+        WhileLoop LessThan(ipos, seqln) {
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                TF_VecSet(qrow, col, TF_MatGet(qvec, ipos, col, dmod))
+                col = Add(col, 1)
+            }
+            jpos = 0
+            WhileLoop LessThan(jpos, seqln) {
+                ds = TF_MatGet(dScores, ipos, jpos, seqln)
+                col = 0
+                WhileLoop LessThan(col, dmod) {
+                    kv     = TF_MatGet(kvec, jpos, col, dmod)
+                    qv     = TF_VecGet(qrow, col)
+                    old_dq = TF_MatGet(dQ,   ipos, col, dmod)
+                    old_dk = TF_MatGet(dK,   jpos, col, dmod)
+                    ndq    = Add(old_dq, Multiply(ds, kv))
+                    ndk    = Add(old_dk, Multiply(ds, qv))
+                    TF_MatSet(dQ, ipos, col, dmod, ndq)
+                    TF_MatSet(dK, jpos, col, dmod, ndk)
+                    col = Add(col, 1)
+                }
+                jpos = Add(jpos, 1)
+            }
+            ipos = Add(ipos, 1)
+        }
+
+        TF_VecFree(qrow, dmod)
+        ReturnValue(1)
+    }
+}

+// =============================================================================
+// dWq dWk dWv and accumulate dEmb
+// Reads from TFBackWA: .Wq .Wk .Wv .emb .seqln .dmod .dK .dV
+// =============================================================================
+Function.TFTrain_BackWeightsA {
+    Input:  dWq:  Integer
+    Input:  dWk:  Integer
+    Input:  dWv:  Integer
+    Input:  dEmb: Integer
+    Input:  dQ:   Integer
+    Output: Integer
+    Body: {
+        Wq    = TFBackWA.Wq
+        Wk    = TFBackWA.Wk
+        Wv    = TFBackWA.Wv
+        emb   = TFBackWA.emb
+        seqln = TFBackWA.seqln
+        dmod  = TFBackWA.dmod
+        dK    = TFBackWA.dK
+        dV    = TFBackWA.dV
+
+        emb_row = TF_VecAlloc(dmod)
+        dq_row  = TF_VecAlloc(dmod)
+        dk_row  = TF_VecAlloc(dmod)
+        dv_row  = TF_VecAlloc(dmod)
+        tmp_row = TF_VecAlloc(dmod)
+        DebugAssert(NotEqual(emb_row, 0), "TFTrain_BackWeightsA: alloc failed")
+
+        pos = 0
+        WhileLoop LessThan(pos, seqln) {
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                TF_VecSet(emb_row, col, TF_MatGet(emb, pos, col, dmod))
+                TF_VecSet(dq_row,  col, TF_MatGet(dQ,  pos, col, dmod))
+                TF_VecSet(dk_row,  col, TF_MatGet(dK,  pos, col, dmod))
+                TF_VecSet(dv_row,  col, TF_MatGet(dV,  pos, col, dmod))
+                col = Add(col, 1)
+            }
+                TF_FOuterAccum(dWq, dq_row, emb_row, dmod, dmod)
+                TF_FOuterAccum(dWk, dk_row, emb_row, dmod, dmod)
+                TF_FOuterAccum(dWv, dv_row, emb_row, dmod, dmod)
+
+            TF_MatTVecMul(tmp_row, Wq, dq_row, dmod, dmod)
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                old = TF_MatGet(dEmb, pos, col, dmod)
+                TF_MatSet(dEmb, pos, col, dmod, Add(old, TF_VecGet(tmp_row, col)))
+                col = Add(col, 1)
+            }
+            TF_MatTVecMul(tmp_row, Wk, dk_row, dmod, dmod)
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                old = TF_MatGet(dEmb, pos, col, dmod)
+                TF_MatSet(dEmb, pos, col, dmod, Add(old, TF_VecGet(tmp_row, col)))
+                col = Add(col, 1)
+            }
+            TF_MatTVecMul(tmp_row, Wv, dv_row, dmod, dmod)
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                old = TF_MatGet(dEmb, pos, col, dmod)
+                TF_MatSet(dEmb, pos, col, dmod, Add(old, TF_VecGet(tmp_row, col)))
+                col = Add(col, 1)
+            }
+            pos = Add(pos, 1)
+        }
+
+        TF_VecFree(emb_row, dmod)
+        TF_VecFree(dq_row,  dmod)
+        TF_VecFree(dk_row,  dmod)
+        TF_VecFree(dv_row,  dmod)
+        TF_VecFree(tmp_row, dmod)
+        ReturnValue(1)
+    }
+}

+// =============================================================================
+// dEmbed and dPosEmb from dEmb
+// Reads from TFBackWB: .tokens .seqln .dmod
+// =============================================================================
+Function.TFTrain_BackWeightsB {
+    Input:  dEmbed:  Integer
+    Input:  dPosEmb: Integer
+    Input:  dEmb:    Integer
+    Output: Integer
+    Body: {
+        tokens = TFBackWB.tokens
+        seqln  = TFBackWB.seqln
+        dmod   = TFBackWB.dmod
+
+        pos = 0
+        WhileLoop LessThan(pos, seqln) {
+            tok_off = Multiply(TFElem.Size, pos)
+            tok     = Dereference(Add(tokens, tok_off))
+            col = 0
+            WhileLoop LessThan(col, dmod) {
+                dv     = TF_MatGet(dEmb,    pos, col, dmod)
+                old_te = TF_MatGet(dEmbed,  tok, col, dmod)
+                old_pe = TF_MatGet(dPosEmb, pos, col, dmod)
+                TF_MatSet(dEmbed,  tok, col, dmod, Add(old_te, dv))
+                TF_MatSet(dPosEmb, pos, col, dmod, Add(old_pe, dv))
+                col = Add(col, 1)
+            }
+            pos = Add(pos, 1)
+        }
+        ReturnValue(1)
+    }
+}

+// =============================================================================
+// ACCURACY
+// =============================================================================
+Function.TFTrain_Accuracy {
+    Input:  probs:   Integer
+    Input:  targets: Integer
+    Input:  seqln:   Integer
+    Input:  vocab:   Integer
+    Output: Integer
+    Body: {
+        correct  = 0
+        prob_row = TF_VecAlloc(vocab)
+        pos = 0
+        WhileLoop LessThan(pos, seqln) {
+            col = 0
+            WhileLoop LessThan(col, vocab) {
+                TF_VecSet(prob_row, col, TF_MatGet(probs, pos, col, vocab))
+                col = Add(col, 1)
+            }
+            pred = TF_VecArgMax(prob_row, vocab)
+            tgt  = TF_VecGet(targets, pos)
+            IfCondition EqualTo(pred, tgt) ThenBlock: {
+                correct = Add(correct, 1)
+            }
+            pos = Add(pos, 1)
+        }
+        TF_VecFree(prob_row, vocab)
+        ReturnValue(correct)
+    }
+}

+// =============================================================================
+// LOSS
+// =============================================================================
+Function.TFTrain_Loss {
+    Input:  probs:   Integer
+    Input:  targets: Integer
+    Input:  seqln:   Integer
+    Input:  vocab:   Integer
+    Output: Integer
+    Body: {
+        prob_row  = TF_VecAlloc(vocab)
+        step_loss = 0
+        pos = 0
+        WhileLoop LessThan(pos, seqln) {
+            col = 0
+            WhileLoop LessThan(col, vocab) {
+                TF_VecSet(prob_row, col, TF_MatGet(probs, pos, col, vocab))
+                col = Add(col, 1)
+            }
+            tgt  = TF_VecGet(targets, pos)
+            loss = TFAct_CrossEntropy(prob_row, tgt)
+            step_loss = Add(step_loss, loss)
+            pos = Add(pos, 1)
+        }
+        avg = Divide(step_loss, seqln)
+        TF_VecFree(prob_row, vocab)
+        ReturnValue(avg)
+    }
